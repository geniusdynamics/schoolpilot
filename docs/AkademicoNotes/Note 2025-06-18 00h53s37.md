CREATE OR REPLACE FUNCTION inv.fn_update_stock_level() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.transaction_type = 'Receive' THEN
        INSERT INTO inv.stock_levels (item_id, warehouse_id, quantity) VALUES (NEW.item_id, NEW.warehouse_id, NEW.quantity)
        ON CONFLICT (item_id, warehouse_id) DO UPDATE SET quantity = inv.stock_levels.quantity + NEW.quantity;
    ELSIF NEW.transaction_type = 'Issue' THEN
        UPDATE inv.stock_levels SET quantity = quantity - NEW.quantity WHERE item_id = NEW.item_id AND warehouse_id = NEW.warehouse_id;
    END IF;
    RETURN NEW;
END;
$$;
CREATE OR REPLACE TRIGGER trg_inv_update_stock_level AFTER INSERT ON inv.stock_transactions FOR EACH ROW EXECUTE FUNCTION inv.fn_update_stock_level();

CREATE OR REPLACE FUNCTION log.fn_log_adm_students_update() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE v_user_id bigint;
BEGIN
    -- Safely get the user ID from the session setting
    BEGIN v_user_id := current_setting('app.current_user_id')::bigint;
    EXCEPTION WHEN OTHERS THEN v_user_id := NULL; END;

    IF OLD.deleted_at IS NULL AND NEW.deleted_at IS NOT NULL THEN
        INSERT INTO log.audit (user_id, action, table_name, record_pk, old_data) VALUES (v_user_id, 'SOFT_DELETE', 'adm.students', OLD.id, jsonb_build_object('admission_no', OLD.admission_no));
    ELSE
        INSERT INTO log.audit (user_id, action, table_name, record_pk, old_data, new_data) VALUES (v_user_id, 'UPDATE', 'adm.students', OLD.id, jsonb_build_object('status', OLD.status, 'current_class_stream_id', OLD.current_class_stream_id), jsonb_build_object('status', NEW.status, 'current_class_stream_id', NEW.current_class_stream_id));
    END IF;
    RETURN NEW;
END;
$$;
CREATE OR REPLACE TRIGGER trg_log_adm_students_after_update AFTER UPDATE ON adm.students FOR EACH ROW EXECUTE FUNCTION log.fn_log_adm_students_update();

DECLARE v_old_history_id bigint;
BEGIN
    PERFORM set_config('app.current_user_id', p_user_id::text, true); -- Set user context for the trigger
    SELECT id INTO v_old_history_id FROM adm.student_class_history WHERE student_id = p_student_id AND end_date IS NULL ORDER BY start_date DESC LIMIT 1;
    IF v_old_history_id IS NOT NULL THEN
        UPDATE adm.student_class_history SET end_date = p_promotion_date WHERE id = v_old_history_id;
    END IF;
    INSERT INTO adm.student_class_history (student_id, class_stream_id, academic_year_id, start_date) VALUES (p_student_id, p_new_class_stream_id, p_academic_year_id, p_promotion_date);
    UPDATE adm.students SET current_class_stream_id = p_new_class_stream_id WHERE id = p_student_id;
END;
$$;
