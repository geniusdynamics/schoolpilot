"""
School ERP Database Structure Analyzer
Parses SQL file and extracts comprehensive database structure information
"""
import re
import json
from collections import defaultdict
from typing import Dict, List, Tuple, Set
import os
class DatabaseAnalyzer:
    def __init__(self, sql_file_path: str):
        self.sql_file_path = sql_file_path
        self.tables = {}
        self.foreign_keys = []
        self.indexes = []
        self.enums = {}
        
    def read_sql_file(self) -> str:
        """Read the SQL file content"""
        with open(self.sql_file_path, 'r', encoding='utf-8') as f:
            return f.read()
    
    def extract_table_structures(self, sql_content: str) -> Dict:
        """Extract all table structures from SQL content"""
        # Pattern to match CREATE TABLE statements
        table_pattern = r'CREATE TABLE.*?`(\w+)`.*?\((.*?)\)\s*ENGINE'
        
        tables = {}
        matches = re.finditer(table_pattern, sql_content, re.DOTALL | re.IGNORECASE)
        
        for match in matches:
            table_name = match.group(1)
            table_definition = match.group(2)
            
            tables[table_name] = self.parse_table_definition(table_definition)
            
        return tables
    
    def parse_table_definition(self, definition: str) -> Dict:
        """Parse individual table definition"""
        table_info = {
            'columns': [],
            'primary_keys': [],
            'foreign_keys': [],
            'unique_keys': [],
            'indexes': [],
            'constraints': []
        }
        
        # Split definition into lines and clean them
        lines = [line.strip() for line in definition.split('\n') if line.strip()]
        
        for line in lines:
            line = line.rstrip(',')
            
            # Skip empty lines and comments
            if not line or line.startswith('--'):
                continue
                
            # Primary key
            if line.upper().startswith('PRIMARY KEY'):
                pk_match = re.search(r'PRIMARY KEY \(`([^`]+)`\)', line)
                if pk_match:
                    table_info['primary_keys'].append(pk_match.group(1))
            
            # Foreign key
            elif line.upper().startswith('CONSTRAINT') and 'FOREIGN KEY' in line.upper():
                fk_info = self.parse_foreign_key(line)
                if fk_info:
                    table_info['foreign_keys'].append(fk_info)
            
            # Unique key
            elif line.upper().startswith('UNIQUE KEY'):
                uk_match = re.search(r'UNIQUE KEY `[^`]*` \(([^)]+)\)', line)
                if uk_match:
                    table_info['unique_keys'].append(uk_match.group(1))
            
            # Regular key/index
            elif line.upper().startswith('KEY'):
                key_match = re.search(r'KEY `[^`]*` \(([^)]+)\)', line)
                if key_match:
                    table_info['indexes'].append(key_match.group(1))
            
            # Column definition
            elif '`' in line and not line.upper().startswith(('PRIMARY', 'UNIQUE', 'KEY', 'CONSTRAINT')):
                column_info = self.parse_column_definition(line)
                if column_info:
                    table_info['columns'].append(column_info)
        
        return table_info
    
    def parse_column_definition(self, line: str) -> Dict:
        """Parse individual column definition"""
        # Extract column name
        col_match = re.match(r'`([^`]+)`\s+(.+)', line)
        if not col_match:
            return None
            
        column_name = col_match.group(1)
        column_spec = col_match.group(2)
        
        # Parse data type
        type_match = re.match(r'(\w+)(?:\(([^)]+)\))?', column_spec)
        data_type = type_match.group(1) if type_match else 'UNKNOWN'
        type_params = type_match.group(2) if type_match and type_match.group(2) else None
        
        # Check for constraints
        is_null = 'NOT NULL' not in column_spec.upper()
        is_auto_increment = 'AUTO_INCREMENT' in column_spec.upper()
        default_value = None
        
        # Extract default value
        default_match = re.search(r'DEFAULT\s+([^,\s]+(?:\s+[^,\s]+)*)', column_spec, re.IGNORECASE)
        if default_match:
            default_value = default_match.group(1)
        
        # Extract comment
        comment_match = re.search(r"COMMENT\s+'([^']*)'", column_spec)
        comment = comment_match.group(1) if comment_match else None
        
        # Check for ENUM values
        enum_values = None
        if data_type.upper() == 'ENUM':
            enum_match = re.search(r"enum\('([^']+(?:','[^']+)*)'\)", column_spec, re.IGNORECASE)
            if enum_match:
                enum_values = [val.strip("'") for val in enum_match.group(1).split("','")]
        
        return {
            'name': column_name,
            'data_type': data_type,
            'type_params': type_params,
            'nullable': is_null,
            'auto_increment': is_auto_increment,
            'default': default_value,
            'comment': comment,
            'enum_values': enum_values
        }
    
    def parse_foreign_key(self, line: str) -> Dict:
        """Parse foreign key constraint"""
        # Pattern: CONSTRAINT `name` FOREIGN KEY (`column`) REFERENCES `table` (`column`)
        fk_pattern = r'CONSTRAINT\s+`([^`]+)`\s+FOREIGN KEY\s+\(`([^`]+)`\)\s+REFERENCES\s+`([^`]+)`\s+\(`([^`]+)`\)(?:\s+(ON DELETE\s+\w+(?:\s+\w+)?))?(?:\s+(ON UPDATE\s+\w+(?:\s+\w+)?))?'
        
        match = re.search(fk_pattern, line, re.IGNORECASE)
        if match:
            return {
                'constraint_name': match.group(1),
                'column': match.group(2),
                'referenced_table': match.group(3),
                'referenced_column': match.group(4),
                'on_delete': match.group(5) if match.group(5) else None,
                'on_update': match.group(6) if match.group(6) else None
            }
        return None
    
    def extract_insert_statements(self, sql_content: str) -> Dict:
        """Extract sample data from INSERT statements"""
        insert_pattern = r'INSERT INTO\s+`(\w+)`.*?VALUES\s*(.*?);'
        inserts = {}
        
        matches = re.finditer(insert_pattern, sql_content, re.DOTALL | re.IGNORECASE)
        for match in matches:
            table_name = match.group(1)
            values_section = match.group(2)
            
            # Count number of INSERT rows for this table
            row_count = len(re.findall(r'\([^)]+\)', values_section))
            inserts[table_name] = row_count
            
        return inserts
    
    def analyze_relationships(self) -> Dict:
        """Analyze table relationships and dependencies"""
        relationships = {
            'parent_tables': defaultdict(list),  # Tables this table references
            'child_tables': defaultdict(list),   # Tables that reference this table
            'relationship_types': {}
        }
        
        for table_name, table_info in self.tables.items():
            for fk in table_info['foreign_keys']:
                # This table references another table
                relationships['parent_tables'][table_name].append({
                    'table': fk['referenced_table'],
                    'column': fk['column'],
                    'referenced_column': fk['referenced_column']
                })
                
                # The referenced table has this table as child
                relationships['child_tables'][fk['referenced_table']].append({
                    'table': table_name,
                    'column': fk['referenced_column'],
                    'referenced_column': fk['column']
                })
        
        return relationships
    
    def generate_statistics(self) -> Dict:
        """Generate database statistics"""
        stats = {
            'total_tables': len(self.tables),
            'total_columns': sum(len(table['columns']) for table in self.tables.values()),
            'total_foreign_keys': sum(len(table['foreign_keys']) for table in self.tables.values()),
            'tables_with_data': 0,
            'tables_by_category': defaultdict(int)
        }
        
        # Categorize tables by prefix
        for table_name in self.tables.keys():
            if table_name.startswith('adm_'):
                stats['tables_by_category']['Administration'] += 1
            elif table_name.startswith('acc_'):
                stats['tables_by_category']['Accounting'] += 1
            elif table_name.startswith('lib_'):
                stats['tables_by_category']['Library'] += 1
            elif table_name.startswith('inv_'):
                stats['tables_by_category']['Inventory'] += 1
            else:
                stats['tables_by_category']['Other'] += 1
        
        return stats
    
    def run_analysis(self) -> Dict:
        """Run complete database analysis"""
        print("Starting database analysis...")
        
        # Read SQL file
        sql_content = self.read_sql_file()
        print(f"Read SQL file with {len(sql_content)} characters")
        
        # Extract table structures
        self.tables = self.extract_table_structures(sql_content)
        print(f"Found {len(self.tables)} tables")
        
        # Extract sample data information
        sample_data = self.extract_insert_statements(sql_content)
        print(f"Found sample data for {len(sample_data)} tables")
        
        # Analyze relationships
        relationships = self.analyze_relationships()
        print("Analyzed table relationships")
        
        # Generate statistics
        stats = self.generate_statistics()
        print("Generated database statistics")
        
        # Compile results
        results = {
            'database_name': 'school_erp_groot',
            'analysis_timestamp': '2025-06-17 16:09:40',
            'statistics': stats,
            'tables': self.tables,
            'relationships': dict(relationships),
            'sample_data': sample_data
        }
        
        return results

    def get_table_relationships(self, table_name: str) -> Dict:
        """Get all relationships for a specific table"""
        relationships = {

            'referenced_by': []  # Tables that reference this table (child tables)
        }
        
        # Get parent relationships
        if table_name in self.relationships.get('parent_tables', {}):
            relationships['references'] = self.relationships['parent_tables'][table_name]
        
        # Get child relationships
        if table_name in self.relationships.get('child_tables', {}):
            relationships['referenced_by'] = self.relationships['child_tables'][table_name]
        
        return relationships
    
    def check_data_integrity(self, table_name: str) -> List[str]:
        """Check for potential data integrity issues"""
        issues = []
        
        if table_name not in self.tables:
            return ['Table not found']
        
        table_info = self.tables[table_name]
        
        # Check for missing primary key
        if not table_info.get('primary_keys'):
            issues.append('No primary key defined')
        
        # Check for foreign keys without proper indexes
        for fk in table_info.get('foreign_keys', []):
            # Check if foreign key column has an index
            fk_column = fk['column']
            has_index = any(fk_column in idx for idx in table_info.get('indexes', []))
            if not has_index:
                issues.append(f'Foreign key {fk_column} lacks index for performance')
        
        # Check for nullable foreign keys that should be required
        for column in table_info.get('columns', []):
            if column['name'].endswith('_id') and column['nullable']:
                # Check if this is actually a foreign key
                is_fk = any(fk['column'] == column['name'] for fk in table_info.get('foreign_keys', []))
                if is_fk:
                    issues.append(f'Foreign key {column["name"]} is nullable - consider making required')
        
        return issues

    def generate_enhanced_views(self):
        """Generate enhanced views for reporting"""
        views = []
        
        # Enhanced student masterlist view
        views.append("""
-- Enhanced Student Master List View
DROP VIEW IF EXISTS `vw_enhanced_student_masterlist`;
CREATE VIEW `vw_enhanced_student_masterlist` AS
SELECT 
    s.id,
    s.student_number,
    CONCAT(s.first_name, ' ', COALESCE(s.middle_name, ''), ' ', s.last_name) AS full_name,
    s.gender,
    s.date_of_birth,
    TIMESTAMPDIFF(YEAR, s.date_of_birth, CURDATE()) AS age,
    s.academic_status,
    s.enrollment_status,
    cs.name AS current_class,
    cst.name AS current_stream,
    CONCAT(ct.first_name, ' ', ct.last_name) AS class_teacher,
    s.nationality,
    s.religion,
    s.emergency_contact_name,
    s.emergency_contact_phone,
    COUNT(DISTINCT pg.id) AS guardian_count,
    COUNT(DISTINCT cm.club_id) AS activity_count,
    s.enrollment_date,
    s.graduation_date,
    s.created_at
FROM adm_students s
LEFT JOIN adm_student_class_history sch ON s.id = sch.student_id 
    AND sch.end_date IS NULL
LEFT JOIN adm_class_streams cst ON sch.class_stream_id = cst.id
LEFT JOIN adm_classes cs ON cst.class_id = cs.id
LEFT JOIN adm_staff ct ON cst.class_teacher_id = ct.id
LEFT JOIN adm_student_guardian_relationships sgr ON s.id = sgr.student_id
LEFT JOIN adm_parents_guardians pg ON sgr.guardian_id = pg.id
LEFT JOIN adm_club_memberships cm ON s.id = cm.student_id
GROUP BY s.id;
""")
        
        # Multi-year financial statement view
        views.append("""
-- Multi-year Student Financial Statement View
DROP VIEW IF EXISTS `vw_student_multiyear_statement`;
CREATE VIEW `vw_student_multiyear_statement` AS
SELECT 
    s.id AS student_id,
    s.student_number,
    CONCAT(s.first_name, ' ', s.last_name) AS student_name,
    ay.name AS academic_year,
    sp.name AS school_period,
    i.vote_head,
    SUM(i.net_amount) AS invoiced_amount,
    COALESCE(SUM(ipa.amount_applied), 0) AS paid_amount,
    (SUM(i.net_amount) - COALESCE(SUM(ipa.amount_applied), 0)) AS balance,
    CASE 
        WHEN (SUM(i.net_amount) - COALESCE(SUM(ipa.amount_applied), 0)) < 0 THEN 'OVERPAID'
        WHEN (SUM(i.net_amount) - COALESCE(SUM(ipa.amount_applied), 0)) = 0 THEN 'CLEARED'
        ELSE 'OUTSTANDING'
    END AS balance_status
FROM adm_students s
CROSS JOIN core_academic_years ay
LEFT JOIN core_school_periods sp ON ay.id = sp.academic_year_id
LEFT JOIN fin_student_invoices i ON s.id = i.student_id 
    AND i.academic_year_id = ay.id 
    AND (i.school_period_id = sp.id OR i.school_period_id IS NULL)
LEFT JOIN fin_invoice_payment_applications ipa ON i.id = ipa.invoice_id
WHERE s.academic_status IN ('Active', 'Graduated', 'Transferred')
GROUP BY s.id, ay.id, sp.id, i.vote_head
HAVING invoiced_amount > 0 OR paid_amount > 0;
""")
        
        # Staff discount summary view
        views.append("""
-- Staff Discount Summary View
DROP VIEW IF EXISTS `vw_staff_discount_summary`;
CREATE VIEW `vw_staff_discount_summary` AS
SELECT 
    st.id AS staff_id,
    st.staff_number,
    CONCAT(st.first_name, ' ', st.last_name) AS staff_name,
    st.employment_type,
    COUNT(DISTINCT sd.student_id) AS students_with_discounts,
    AVG(sd.discount_percentage) AS avg_discount_percentage,
    SUM(CASE WHEN sd.is_active = 1 THEN 1 ELSE 0 END) AS active_discounts,
    GROUP_CONCAT(DISTINCT CONCAT(s.first_name, ' ', s.last_name) SEPARATOR '; ') AS discounted_students
FROM adm_staff st
LEFT JOIN fin_staff_discounts sd ON st.id = sd.staff_id
LEFT JOIN adm_students s ON sd.student_id = s.id
GROUP BY st.id;
""")

        # Focus on core tables for clarity

-- Validation Query 1: Student Count by Academic Status
SELECT 
    academic_status,
    COUNT(*) as student_count,
    ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM adm_students), 2) as percentage
FROM adm_students 
GROUP BY academic_status
ORDER BY student_count DESC;
-- Validation Query 2: Staff Distribution by Employment Type
SELECT 
    employment_type,
    COUNT(*) as staff_count,
    ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM adm_staff), 2) as percentage
FROM adm_staff 
GROUP BY employment_type
ORDER BY staff_count DESC;
-- Validation Query 3: Family Structures (Sibling Relationships)
SELECT 
    'Students with Siblings' as category,
    COUNT(DISTINCT student_id_1) + COUNT(DISTINCT student_id_2) as count
FROM adm_student_siblings
UNION ALL
SELECT 
    'Total Sibling Pairs' as category,
    COUNT(*) as count
FROM adm_student_siblings;
-- Validation Query 4: Financial Summary by Vote Head
SELECT 
    vote_head,
    COUNT(*) as invoice_count,
    SUM(total_amount) as total_invoiced,
    SUM(discount_amount) as total_discounts,
    SUM(net_amount) as total_net_amount
FROM fin_student_invoices 
GROUP BY vote_head
ORDER BY total_net_amount DESC;
-- Validation Query 5: Staff Discount Coverage
SELECT 
    sd.discount_percentage,
    COUNT(*) as students_count,
    COUNT(DISTINCT sd.staff_id) as staff_count
FROM fin_staff_discounts sd
WHERE sd.is_active = 1
GROUP BY sd.discount_percentage
ORDER BY sd.discount_percentage DESC;
-- Validation Query 6: Club Membership Distribution
SELECT 
    c.name as club_name,
    c.type as club_type,
    COUNT(cm.student_id) as member_count
FROM adm_clubs_and_societies c
LEFT JOIN adm_club_memberships cm ON c.id = cm.club_id
GROUP BY c.id, c.name, c.type
ORDER BY member_count DESC;

    grade,
    COUNT(*) as result_count,
    ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM edm_student_exam_results), 2) as percentage
FROM edm_student_exam_results
GROUP BY grade
ORDER BY grade;
-- Validation Query 8: Library Usage Statistics
SELECT 
    status,
    COUNT(*) as loan_count,
    ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM lib_loans), 2) as percentage
FROM lib_loans
GROUP BY status
ORDER BY loan_count DESC;

FROM fin_student_payments

-- Library usage statistics by student
SELECT s.student_number, s.first_name, s.last_name,
       COUNT(ll.id) as books_borrowed,
       SUM(CASE WHEN ll.status = 'Returned' THEN 1 ELSE 0 END) as books_returned,
