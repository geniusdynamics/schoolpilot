+const fs = require('fs');
+const code = fs.readFileSync('/workspace/src/contexts/AppDataContext.tsx', 'utf8');
+
+let parenCount = 0;
+let braceCount = 0;
+let bracketCount = 0;
+
+const lines = code.split('\n');
+
+for (let i = 0; i < lines.length; i++) {
+  const line = lines[i];
+  for (let j = 0; j < line.length; j++) {
+    const c = line[j];
+    if (c === '(') parenCount++;
+    else if (c === ')') parenCount--;
+    else if (c === '{') braceCount++;
+    else if (c === '}') braceCount--;
+    else if (c === '[') bracketCount++;
+    else if (c === ']') bracketCount--;
+  }
+  
+  if (parenCount < 0 || braceCount < 0 || bracketCount < 0) {
+    console.log(`Mismatch at line ${i + 1}: parens=${parenCount}, braces=${braceCount}, brackets=${bracketCount}`);
+  }
+}
+
+console.log(`Final count: parens=${parenCount}, braces=${braceCount}, brackets=${bracketCount}`);

+import React, { createContext, useContext, useCallback, ReactNode } from 'react';

+export interface PaginationOptions {
+  page: number;
+  pageSize: number;
+  sortBy?: string;
+  sortOrder?: 'asc' | 'desc';
+}
+
+export interface PaginationResult<T> {
+  data: T[];
+  page: number;
+  pageSize: number;
+  totalCount: number;
+  totalPages: number;
+  hasNextPage: boolean;
+  hasPreviousPage: boolean;
+}
+
+export interface PaginationContextType {
+  getCollectionPage: <T = any>(
+    collectionName: string,
+    options: PaginationOptions
+  ) => Promise<PaginationResult<T>>;
+}
+
+export const PaginationContext = createContext<PaginationContextType | undefined>(undefined);
+
+export const PaginationProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
+  const getCollectionPage = useCallback(async <T = any,>(
+    collectionName: string,
+    options: PaginationOptions
+  ): Promise<PaginationResult<T>> => {
+    try {
+      const dbManager = UnifiedDatabaseManager.getInstance();
+      const collection = dbManager.getCollection<T>(collectionName);
+      if (!collection) {
+        throw new Error(`Collection ${collectionName} not found`);
+      }
+
+      const totalCount = await collection.count();
+      const totalPages = Math.ceil(totalCount / options.pageSize);
+      const skip = (options.page - 1) * options.pageSize;
+
+      let query = collection.find();
+
+      if (options.sortBy) {
+        query = query.sort({
+          [options.sortBy]: options.sortOrder || 'asc'
+        });
+      }
+
+      query = query.skip(skip).limit(options.pageSize);
+
+      const docs = await query.exec();
+      const data = docs.map(doc => doc.toJSON() as T);
+
+      return {
+        data,
+        page: options.page,
+        pageSize: options.pageSize,
+        totalCount,
+        totalPages,
+        hasNextPage: options.page < totalPages,
+        hasPreviousPage: options.page > 1
+      };
+    } catch (err) {
+      console.error(`[PaginationContext] Error fetching page for ${collectionName}:`, err);
+      throw err;
+    }
+  }, []);
+
+  return (
+    <PaginationContext.Provider value={{ getCollectionPage }}>
+      {children}
+    </PaginationContext.Provider>
+  );
+};

+                  ids.add(doc.id);
+                }
+              }
+            }
+
+            if (duplicates.length > 0) {
+              issues.push({
+                severity: 'error',
+                message: `Collection ${name} has ${duplicates.length} duplicate IDs`,
+                collection: name
+              });
+            }
+          }
+        } catch (err) {
+          issues.push({
+            severity: 'warning',
+            message: `Failed to validate collection ${name}: ${err instanceof Error ? err.message : String(err)}`,
+            collection: name
+          });
+        }
+      }
+
+      const isValid = !issues.some(issue => issue.severity === 'error');
+      
+      if (isValid) {
+        toast.showSuccess('Data validation passed');
+      } else {
+        toast.showWarning(`Data validation found ${issues.filter(i => i.severity === 'error').length} errors`);
+      }
+
+      console.log(`✅ [ValidationContext] Data validation completed. Found ${issues.length} issues.`);
+      return { isValid, issues };
+    } catch (err) {
+      console.error('❌ [ValidationContext] Global validation error:', err);
+      throw err;
+    }
+  }, [toast]);
+
+  return (
+    <ValidationContext.Provider value={{ validateData }}>
+      {children}
+    </ValidationContext.Provider>
+  );
+};

+export function createDomainContext<T>(domainName: string, collectionsList: string[]) {
+  const DomainContext = createContext<T | undefined>(undefined);
+
+  const DomainProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
+    const [data, setData] = useState<Partial<T>>({});
+    
+    const loadDomainCollections = useCallback(async () => {
+      try {
+        const dbManager = UnifiedDatabaseManager.getInstance();
+        const loadedData: Record<string, any[]> = {};
+
+        for (const name of collectionsList) {
+          try {
+            const collection = dbManager.getCollection(name);
+            if (collection && typeof collection.find === 'function') {
+              const docs = await collection.find({});
+              loadedData[name] = docs;
+            } else {
+              loadedData[name] = [];
+            }
+          } catch (err) {
+            console.warn(`[${domainName}] Failed to load collection ${name}:`, err);
+            loadedData[name] = [];
+          }
+        }
+
+        // We normalize the fetched data to get the camelCase versions
+        const normalized = normalizeAppDataCollections(loadedData) as any;
+        
+        // Filter out keys that we didn't request
+        const domainData: any = {};
+        
+        // helper to get camelCase mapping just like normalizeAppDataCollections does
+        const snakeToCamel = (value: string): string => value.replace(/_([a-z])/g, (_, letter: string) => letter.toUpperCase());
+        
+        // Include aliases mapping if available (we'll just manually check the normalized output)
+        // Since normalizeAppDataCollections populated the camelCase keys correctly for loadedData
+        // We just need to identify which camelCase keys correspond to our snake_case collectionsList
+        
+        const camelKeys = collectionsList.map(name => {
+           if (name === 'app_settings' || name === 'school_details') return 'schoolDetails';
+           return snakeToCamel(name);
+        });
+
+        for (const key of camelKeys) {
+          if (normalized[key] !== undefined) {
+             domainData[key] = normalized[key];
+          }
+        }
+
+        setData(domainData);
+      } catch (err) {
+        console.error(`[${domainName}] Failed to load collections:`, err);
+      }
+    }, []);
+
+    useEffect(() => {
+      loadDomainCollections();
+    }, [loadDomainCollections]);
+
+    useEffect(() => {
+      const dataModeManager = getDataModeManager();
+      
+      const handleDataChanged = () => {
+        console.log(`🔄 [${domainName}] Received appDataChanged event, reloading collections...`);
+        loadDomainCollections();
+      };
+      
+      window.addEventListener('appDataChanged', handleDataChanged);
+      dataModeManager.on('dataSeeded', handleDataChanged);
+      
+      return () => {
+        window.removeEventListener('appDataChanged', handleDataChanged);
+        dataModeManager.off('dataSeeded', handleDataChanged);
+      };
+    }, [loadDomainCollections]);
+
+    return (
+      <DomainContext.Provider value={data as T}>
+        {children}
+      </DomainContext.Provider>
+    );
+  };
+
+  const useDomainContext = () => {
+    const context = useContext(DomainContext);
+    if (context === undefined) {
+      throw new Error(`use${domainName}Context must be used within a ${domainName}Provider`);
+    }
+    return context;
+  };
+
+  return { DomainContext, DomainProvider, useDomainContext };
+}
